extends RefCounted
## Authored shape study using exactly the runtime direct-sculpt brush API.
## Deliberate strokes, not automatic rock placement. Not a default preset yet.
static func author(data)->void:
 data.begin_stroke()
 data.apply_rock_brush(Vector3(-15.0,2.1,4.25),2.65,'rock_add')
 data.apply_rock_brush(Vector3(-12.6,1.8,4.85),2.1,'rock_add')
 data.apply_rock_brush(Vector3(-15.45,3.2,4.15),1.9,'rock_add')
 data.apply_rock_brush(Vector3(-15.0,4.7,4.3),4,'rock_plane',Vector3(.18,1,.10))
 data.apply_rock_brush(Vector3(-14.6,2.3,6.18),4,'rock_plane',Vector3(-.18,.08,1))
 data.apply_rock_brush(Vector3(-17.1,2.6,4.3),4,'rock_plane',Vector3(-1,.04,.25))
 data.apply_rock_brush(Vector3(-11.0,2.1,5.0),4,'rock_plane',Vector3(1,.03,.22))
 data.apply_rock_brush(Vector3(-12.1,3.25,5.4),2.2,'rock_plane',Vector3(.65,.75,.30))
 data.apply_rock_brush(Vector3(-13.5,3.6,6.0),.42,'rock_cut')
 data.end_stroke()
