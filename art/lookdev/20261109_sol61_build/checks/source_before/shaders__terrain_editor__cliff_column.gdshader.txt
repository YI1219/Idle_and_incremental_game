shader_type spatial;
#include "terrain_morph.gdshaderinc"
#include "cliff_palette.gdshaderinc"
#include "stone_surface.gdshaderinc"
// Multi-Boulder Stacked & Staggered Chiseled Shanshui Cliff Shader (纵横交错叠石 · 干脆切削块面 · 零掠射阴影锯齿, Refs 01–04).
uniform float crown_cover = 1.0;
varying vec3 local_position;
varying vec3 column_origin;
varying vec3 world_position;
varying vec3 world_facet_n;
varying float v_world_ny;

void vertex(){
	local_position=VERTEX;
	column_origin=(MODEL_MATRIX*vec4(0.0,0.0,0.0,1.0)).xyz;
	world_position=(MODEL_MATRIX*vec4(VERTEX,1.0)).xyz;
	world_facet_n=normalize(MODEL_NORMAL_MATRIX*NORMAL);
	v_world_ny=world_facet_n.y;
	if(terrain_morph_progress<1.0){
		vec3 origin=(MODEL_MATRIX*vec4(0.0,0.0,0.0,1.0)).xyz;
		VERTEX+=(inverse(MODEL_MATRIX)*vec4(0.0,terrain_morph_offset(origin),0.0,0.0)).xyz;
	}
}

void fragment(){
	// 1. Crisp, flat-planar chiseled boulder faces with delicate mineral paper tooth (zero wavy bumps!):
	vec3 base_normal=normalize(world_facet_n);
	vec3 detail_normal=stone_normal(world_position*1.80,base_normal);
	vec3 shade_normal=normalize(mix(base_normal,detail_normal,.035));
	NORMAL=normalize((VIEW_MATRIX*vec4(base_normal,0.0)).xyz);

	// 2. Per-boulder & per-sub-block mineral value/temperature separation (4A: clear boulder-to-boulder identity):
	vec3 stone=stone_surface(world_position,shade_normal)*mix(vec3(1.0),COLOR.rgb/.708,.84);
	// Broad mineral washes remain subordinate to the authored planar silhouette.
	float wash_luma=dot(stone_gouache(world_position,base_normal),vec3(.299,.587,.114));
	float mineral_luma=dot(stone_mineral(world_position*.72,base_normal),vec3(.299,.587,.114));
	float wash_value=clamp(mix(1.0,wash_luma/.50,.75),.88,1.10);
	float mineral_value=clamp(mix(1.0,mineral_luma/.34,.16),.94,1.06);
	stone*=wash_value*mineral_value;

	// Key and fill directions are evaluated only in light(), without a baked sun vector.

	// 4. Clean, uniform gouache planes (zero artificial local_position.z / UV.y dark bands across flat faces!):
	float top_gate=(crown_cover>.5) ? smoothstep(-.03,-.14,local_position.y) : 1.0;
	float top_shelf=smoothstep(.46,.84,shade_normal.y)*top_gate;
	float under_belly=smoothstep(-.22,-.72,base_normal.y)*(1.0-top_shelf);
	float ao_val=mix(1.0,.91,under_belly);
	ao_val=mix(ao_val,1.0,top_shelf*.92);

	float lip_ao=stone_lip_ao(world_position,base_normal,column_origin.y,crown_cover);
	ao_val*=mix(1.0,.46,lip_ao);
	stone=stone_root_contact(stone,world_position,base_normal,column_origin.y,crown_cover);
	ALBEDO=stone_mist_dissolve(stone,world_position,base_normal);
	ROUGHNESS=.952;
	SPECULAR=.016;
	AO=ao_val;
	AO_LIGHT_AFFECT=.76;
}

void light(){
	vec3 lc=LIGHT_COLOR/PI;
	float energy=dot(lc,vec3(.299,.587,.114));
	float ndl=dot(NORMAL,LIGHT);
	if(LIGHT_IS_DIRECTIONAL && max(lc.r,max(lc.g,lc.b))>.45){
		// Continuous matte response keeps the root part of the same rock mass.
		// The previous narrow terminator turned every short foot into a black cuff.
		float direct=max(ndl,0.0)*.86*ATTENUATION;
		float sky_weight=smoothstep(-.80,.90,v_world_ny);
		vec3 sky=vec3(.62,.77,1.08)*(.095+.075*sky_weight);
		vec3 cloud=vec3(.88,.85,.76)*(.055*(1.0-sky_weight));
		DIFFUSE_LIGHT+=lc*direct+(sky+cloud)*energy;
	}else{
		stone_crisp_facet_light(NORMAL,LIGHT,LIGHT_COLOR,ATTENUATION,LIGHT_IS_DIRECTIONAL,v_world_ny,DIFFUSE_LIGHT);
	}
}
