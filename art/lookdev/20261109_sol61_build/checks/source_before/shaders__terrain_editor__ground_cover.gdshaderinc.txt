#include "weathering.gdshaderinc"
uniform vec2 footpath_points[8];
uniform int footpath_count=0;
uniform sampler2D ground_detail_map : filter_linear_mipmap_anisotropic, repeat_enable;
// Fine colour and lighting contrast. Lower default yields soft color-block cover.
uniform float ground_fine_strength : hint_range(0.0, 1.0) = .05;
uniform float lantern_glow_strength : hint_range(0.0, 1.0) = 0.0;
uniform vec3 lantern_glow_color : source_color = vec3(1.0, .65, .25);
// World-space compression is supplied by TerrainRenderer.set_grass_compression.
uniform sampler2D grass_compression : filter_linear, repeat_disable, hint_default_black;
uniform vec2 compression_origin = vec2(-64.0);
uniform vec2 compression_extent = vec2(128.0);
float gc_standing(vec2 p){
 vec2 uv=(p-compression_origin)/compression_extent;
 float inside=step(0.0,uv.x)*step(0.0,uv.y)*step(uv.x,1.0)*step(uv.y,1.0);
 return 1.0-texture(grass_compression,uv).r*inside;
}
float gc_hash(vec2 p){return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453);}
vec2 gc_hash2(vec2 p){return fract(sin(vec2(dot(p,vec2(127.1,311.7)),dot(p,vec2(269.5,183.3))))*43758.5453);}
float gc_noise(vec2 p){
	vec2 i=floor(p);vec2 f=fract(p);
	// C2 quintic interpolation eliminates grid-line derivative creases and moire bands when zoomed in:
	vec2 u=f*f*f*(f*(f*6.0-15.0)+10.0);
	return mix(mix(gc_hash(i),gc_hash(i+vec2(1,0)),u.x),mix(gc_hash(i+vec2(0,1)),gc_hash(i+vec2(1,1)),u.x),u.y);
}
float gc_cover_height(vec2 p){
	mat2 turn=mat2(vec2(.8,.6),vec2(-.6,.8));
	return gc_noise(p*1.1)*.56+gc_noise(turn*p*2.3+vec2(5.2,1.3))*.30+gc_noise(turn*turn*p*4.2+vec2(2.7,8.1))*.14;
}
float gc_path_dist(vec2 p){
	float distance_to_path=1000.0;
	for(int i=0;i<7;i++){
		if(i+1>=footpath_count)break;
		vec2 a=footpath_points[i];vec2 edge=footpath_points[i+1]-a;
		float t=clamp(dot(p-a,edge)/max(dot(edge,edge),.0001),0.0,1.0);
		distance_to_path=min(distance_to_path,length(p-a-edge*t));
	}
	return distance_to_path;
}
// Broad 1.6m–4.2m isotropic soft moss-meadow cushion height (rotated octaves, zero grid or stripe artifacts!):
float gc_pillow_core(vec2 p){
	mat2 r1=mat2(vec2(.80,.60),vec2(-.60,.80));
	mat2 r2=mat2(vec2(-.44,.90),vec2(-.90,-.44));
	return gc_noise(r1*p*.34+vec2(3.7,8.2))*.62+gc_noise(r2*p*.68+vec2(14.3,5.9))*.38;
}
float gc_pillow_h(vec2 p){
	float d_path=gc_path_dist(p);
	float path_shoulder=smoothstep(.48,.92,d_path)*(1.0-smoothstep(1.05,1.85,d_path))*.14;
	return gc_pillow_core(p)+path_shoulder;
}
vec2 gc_pillow_grad(vec2 p){
	float e=.22;
	float h0=gc_pillow_core(p);
	return vec2(gc_pillow_core(p+vec2(e,0.0))-h0,gc_pillow_core(p+vec2(0.0,e))-h0)/e;
}
// Multi-scale gouache meadow cushions (水粉绒毯草甸 · 柔和色块 · 零条纹零摩尔纹):
vec3 ground_cover(vec3 position,float detail,out vec2 slope){
	vec2 p=position.xz;
	float standing=gc_standing(p);
	detail*=standing;
	float fine_detail=detail*ground_fine_strength;
	mat2 rot_a=mat2(vec2(.80,.60),vec2(-.60,.80));
	mat2 rot_b=mat2(vec2(-.44,.90),vec2(-.90,-.44));

	// 1. Macro (4–10m): wet-on-wet watercolor clearings (sunlit spring-tea vs cool celadon-moss)
	float wash_a=gc_noise(rot_a*p*.20+vec2(4.3,9.1));
	float wash_b=gc_noise(rot_b*p*.42+vec2(13.7,2.4));
	float wash_c=gc_noise(p*.12+vec2(7.8,3.2));
	vec3 base=mix(grass_base,grass_highlight,clamp(wash_c*.52+wash_a*.48,0.0,1.0));

	float cover=smoothstep(.22,.78,wash_a*.58+wash_b*.42);
	float cloud_dapple=gc_noise(p*.085+vec2(TIME*.012,-TIME*.008));
	float height=gc_cover_height(p);
	float pillow=gc_pillow_h(p);
	vec3 sunlit_tea=base*vec3(1.18,1.14,.88);
	vec3 celadon_moss=base*vec3(.84,.98,.78);
	vec3 color=mix(sunlit_tea,celadon_moss,clamp(cover*.58+(1.0-pillow)*.14+(height-.5)*.6*fine_detail,0.0,1.0));
	float sun_sweep=smoothstep(-19.0,17.0,p.x*.88+p.y*.22);
	color*=mix(vec3(.96,.99,1.02),vec3(1.09,1.06,.95),sun_sweep);
	color*=mix(.92,1.09,smoothstep(.16,.84,cloud_dapple));

	// 2. Meso (1.4m–2.8m): broad, soft gouache moss-and-turf color blocks (isotropic, zero small polka-dots or stripes!)
	vec2 slope_tuft=vec2(0.0);
	vec2 g=rot_a*p*.56+vec2(17.3,5.1);
	vec2 cell=floor(g);vec2 f=fract(g);
	float best=0.0;vec2 lean=vec2(0.0);float tone=0.0;
	for(int y=-1;y<=1;y++)for(int x=-1;x<=1;x++){
		vec2 o=vec2(float(x),float(y));
		vec2 id=cell+o;
		float present=step(gc_hash(id+vec2(9.0,2.0)),.66+.22*cover);
		vec2 d=o+gc_hash2(id)*.74+.13-f;
		float r=mix(.54,.86,gc_hash(id+vec2(4.0,9.0)));
		float tuft=(1.0-smoothstep(0.0,r,length(d)))*present;
		if(tuft>best){best=tuft;lean=-d/r;tone=gc_hash(id+vec2(1.0,7.0));}
	}
	vec3 cushion_cool=color*vec3(.87,.96,.82);
	vec3 cushion_warm=color*vec3(1.12,1.10,.93);
	vec3 tuft_color=mix(cushion_cool,cushion_warm,smoothstep(.10,.90,tone))*(.96+.08*best);
	color=mix(color,tuft_color,smoothstep(0.0,.52,best)*.32*standing);
	slope_tuft-=lean*smoothstep(0.0,.6,best)*.18;

	// 3. Soft isotropic watercolor paper grain (strictly gated by fine_detail = 0.05; zero directional corduroy stripes!):
	float soft_grain=gc_noise(rot_b*p*1.35+vec2(6.1,11.4))*.60+gc_noise(rot_a*p*2.65+vec2(19.2,4.7))*.40;
	vec3 grain_tint=mix(vec3(.96,.98,.95),vec3(1.04,1.03,.98),soft_grain);
	color*=mix(vec3(1.0),grain_tint,fine_detail*2.2);

	float e=.08;
	slope=vec2(gc_cover_height(p+vec2(e,0.0))-height,gc_cover_height(p+vec2(0.0,e))-height)/e;
	slope=(slope*.04+slope_tuft*.65)*standing*ground_fine_strength;
	vec4 mapped=texture(ground_detail_map,p*.36);
	color*=1.0+(mapped.r-.5)*.14*fine_detail;
	slope+=(mapped.gb-.5)*.8*standing*ground_fine_strength;

	// Crisp, warm earth-sandstone footpath bed hugging the stepping stones (zero dusty grey halo bleeding into lawn!):
	float distance_to_path=gc_path_dist(p);
	float soft_wobble=(gc_noise(p*1.6)-.5)*.10+(gc_noise(p*4.2)-.5)*.04;
	float wear=1.0-smoothstep(.24+soft_wobble,.52+soft_wobble,distance_to_path);
	float moss_verge=smoothstep(.34,.52,distance_to_path)*(1.0-smoothstep(.54,.84,distance_to_path));
	color=mix(color,color*vec3(.88,1.02,.84),moss_verge*.28);
	float soil=max(wear*.92,(1.0-standing)*.42);
	float sand_wash=gc_noise(p*1.1)*.65+gc_noise(p*2.8)*.35;
	vec3 warm_sand=mix(vec3(.245,.236,.145),vec3(.330,.312,.192),sand_wash);
	color=mix(color,warm_sand,soil);
	slope*=1.0-soil*.85;

	if(lantern_glow_strength>0.001){
		float d1=length(p-vec2(12.3,-2.45));
		float glow=exp(-d1*d1*.68)*lantern_glow_strength;
		color+=lantern_glow_color*glow*.42;
	}
	return color;
}
